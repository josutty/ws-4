# BUG-001: Successful login does not navigate to dealer home

## Report

Steps: open `/login`, enter `DLR-1002` and `REDACTED`, submit **Sign in**.

Expected: successful authentication updates the session and navigates to `/dealer/home`.

Actual: the login request succeeds, but the browser remains on `/login` and the session remains signed out.

## Root cause (from debugger)

`src/features/authenticate/ui/SignInForm.tsx:38-53` awaits `loginDealer(...).unwrap()` but the success path does not dispatch the existing `signedIn` session action or call navigation. The same omission exists in the SSO success path at lines 55-63. `src/entities/session/model/sessionSlice.ts:16-20` provides `signedIn`, and `routes.dealerHome` is `/dealer/home`.

```tsx
// test/test-utils.tsx
import type { ReactElement, ReactNode } from 'react';
import { render, type RenderOptions } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { Provider } from 'react-redux';
import { MemoryRouter } from 'react-router';
import { makeStore } from '@app/store';

interface ProviderOptions extends Omit<RenderOptions, 'wrapper'> {
  preloadedState?: Parameters<typeof makeStore>[0];
  store?: ReturnType<typeof makeStore>;
  route?: string;
}

/** Fresh store per call (AGENTS.md §5). Returns the store and a user-event instance. */
export function renderWithProviders(ui: ReactElement, { preloadedState, store = makeStore(preloadedState), route = '/', ...options }: ProviderOptions = {}) {
  function Wrapper({ children }: { children: ReactNode }) {
    return (
      <Provider store={store}>
        <MemoryRouter initialEntries={[route]}>{children}</MemoryRouter>
      </Provider>
    );
  }
  return { store, user: userEvent.setup(), ...render(ui, { wrapper: Wrapper, ...options }) };
}

export * from '@testing-library/react';
```
